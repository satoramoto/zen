export const useTheme = () => {
  const [theme, setTheme] = useState(currentTheme)
  useEffect(() => onTheme(setTheme), [])
  return theme
}

export const Themed = ({ children }) => (
  <div data-theme={useTheme()}>{children}</div>
)

export const tokens = (css, prefix = '') =>
  [...css.matchAll(/^\s*(--[\w-]+)\s*:\s*([^;]+);(?:[ \t]*\/\*\s*(.*?)\s*\*\/)?/gm)]
    .map((m) => ({ name: m[1], value: m[2].trim(), note: m[3] ?? '' }))
    .filter((t) => t.name.startsWith(prefix))

export const grid = { display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(168px, 1fr))', gap: 'var(--space-12)', margin: 'var(--space-16) 0' }
export const label = { font: 'var(--type-text)', letterSpacing: 'var(--tracking-text)', color: 'var(--caption-ink)' }
export const name = { font: 'var(--type-text)', letterSpacing: 'var(--tracking-text)', fontFamily: 'var(--font-mono)', color: 'var(--t)' }

export const isAmount = (value) =>
  // A number or percentage (an opacity, a mix amount): listed by Amounts, not drawn as a swatch.
  /^-?[\d.]+(%|ms)?$/.test(value)

export const Amounts = ({ css, names }) => {
  // Tokens listed by value: the amounts in `css`, or the tokens of `css` named in `names`.
  const overrides = useTheme() === 'light' ? Object.fromEntries(tokens(light).map((t) => [t.name, t.value])) : {}
  return (
  <div style={{ margin: 'var(--space-16) 0' }}>
    {tokens(css)
      .filter((t) => (names ? names.includes(t.name) : isAmount(t.value)))
      .map((t) => (
        <div key={t.name} style={{ display: 'flex', alignItems: 'baseline', gap: 'var(--space-16)' }}>
          <span style={{ ...name, width: 200, flex: 'none' }}>{t.name}</span>
          <span style={label}>{overrides[t.name] ?? t.value}{t.note ? ` · ${t.note}` : ''}</span>
        </div>
      ))}
  </div>
  )
}

export const Swatches = ({ css, glows = false }) => {
  const overrides = useTheme() === 'light' ? Object.fromEntries(tokens(light).map((t) => [t.name, t.value])) : {}
  return (
  <div style={grid}>
    {tokens(css)
      .filter((t) => !isAmount(t.value))
      .filter((t) => glows === /px/.test(t.value))
      .map((t) => ({ ...t, value: overrides[t.name] ?? t.value, note: overrides[t.name] ? '' : t.note }))
      .map((t) => (
        <div key={t.name}>
          <div
            style={{
              height: 40,
              borderRadius: 'var(--radius)',
              border: 'var(--line-width) solid var(--line)',
              background: glows ? 'var(--btn)' : `var(${t.name})`,
              boxShadow: glows ? `var(${t.name})` : undefined,
            }}
          />
          <div style={name}>{t.name}</div>
          <div style={label}>{t.value}{t.note ? ` · ${t.note}` : ''}</div>
        </div>
      ))}
  </div>
  )
}

export const face = (state, hue) => {
  const base = { position: 'relative', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', width: 88, height: 'var(--control-height)', font: 'var(--type-text)', letterSpacing: 'var(--tracking-text)', isolation: 'isolate' }
  if (state === 'on') return { ...base, background: `var(${hue})`, color: 'var(--on-ink)' }
  if (state === 'waiting') return { ...base, boxShadow: `inset 0 0 0 var(--outline-width-wait) var(${hue})`, color: `var(${hue})` }
  // Absent: the control's own hue at --absent-mix over the ground (tokens/state.css), never grey.
  if (state === 'absent') return { ...base, boxShadow: `inset 0 0 0 var(--outline-width) var(--absent-${hue.slice(2)})`, color: `var(--absent-${hue.slice(2)})` }
  return { ...base, boxShadow: `inset 0 0 0 var(--outline-width) var(${hue})`, color: `var(${hue})` }
}

# Tokens

Every token the UI library has, in the theme the toolbar shows: flip the theme to review the other one. Components use only the **roles**; the **palette** is what the themes map the roles to (`src/ui/tokens/`, see `docs/factory/storybook-axioms.md`, axioms 2 and 5).

## Roles

The Push canvas's own names. A swatch is the role's colour in the current theme.

<Swatches css={dark} />

### Glows

`none` in light.

<Swatches css={dark} glows />

### Amounts

Opacities and mixes the roles apply to a colour.

<Amounts css={dark} />

## Palette

The raw colours. Only the theme files read them.

<Swatches css={palette} />

## Type

Five roles. One line or row uses one size: a label and its value differ by colour and weight, never by size; only the display's hero values are large. A component sets a role's `font` and its `--tracking-*`, nothing else.

<Type />

## State

One treatment per state for every control. Rest: a 1px outline and label in the control's hue (`--neutral` when it has none). On, chosen, playing: a solid fill in the hue under `--on-ink`. Waiting: a 2px ring over a faint fill. Disabled or absent: the control's own hue at `--absent-mix` strength over the ground, outline and label, so an absent Intro pad reads as a faded yellow pad; never grey. Each hue has its token, `--absent-<hue>` (`tokens/state.css`); `--absent` is the neutral one. Tabs and One Touch numbers have no rest outline, only the chosen fill.
